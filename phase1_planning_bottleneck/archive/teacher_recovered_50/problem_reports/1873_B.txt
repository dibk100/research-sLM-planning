def max_product_after_adding_one(digits):
    n = len(digits)
    max_product = 0
    
    # Iterate through each digit
    for i in range(n):
        # Calculate the new product if the current digit is increased by 1
        new_product = (digits[i] + 1) * (10**(n-1-i))
        
        # Update the maximum product if the new product is greater
        if new_product > max_product:

    return max_product

def max_product_after_increment(a):
    n = len(a)
    max_product = 0
    
    # Try incrementing each digit
    for i in range(n):
        # Create a new array with the digit incremented
        new_a = a[:]
        new_a[i] += 1
        
        # Calculate the product of the new array
        product = 1
        for num in new_a:
            product *= num
        
        # Update the maximum product found
        max_product = max(max_product, product)
    
    return max_product
