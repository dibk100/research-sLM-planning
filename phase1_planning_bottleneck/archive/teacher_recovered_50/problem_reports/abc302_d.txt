import sys
from typing import List

def binary_search(arr: List[int], target: int) -> int:
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] <= target:
            left = mid + 1
        else:
            right = mid - 1
    return right

def main():
    input = sys.stdin.read
    data = input().split()
    
    N, M, D = map(int, data[:3])
    A = list(map(int, data[3:3+N]))
    B = list(map(int, data[3+N:3+N+M]))
    
    B.sort()
    
    max_sum = -1
    
    for a in A:
        # Find the largest b such that b <= a + D and b >= a - D
        b_index = binary_search(B, a + D)
        if b_index >= 0 and B[b_index] >= a - D:
            max_sum = max(max_sum, a + B[b_index])
    
    print(max_sum)

if __name__ == "__main__":
    main()
