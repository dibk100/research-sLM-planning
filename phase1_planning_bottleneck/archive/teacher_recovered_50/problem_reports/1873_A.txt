import sys

def can_sort_with_one_swap(s):
    # Check if the string is already sorted in the desired order
    if s == 'abc' or s == 'acb' or s == 'bac':
        return "YES"
    
    # Check if there is exactly one pair of adjacent characters that can be swapped
    count = 0
    for i in range(len(s) - 1):
        if s[i] > s[i + 1]:
            count += 1
    
    # If there is exactly one such pair, output "YES"
    if count == 1:
        return "YES"
    
    # Otherwise, output "NO"
    return "NO"

# Read input
t = int(sys.stdin.readline())
for _ in range(t):
    s = sys.stdin.readline().strip()
    print(can_sort_with_one_swap(s))
