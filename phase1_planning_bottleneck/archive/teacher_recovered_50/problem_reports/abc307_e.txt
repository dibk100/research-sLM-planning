def distinct_adjacent(n, m):
    MOD = 998244353
    
    # Base cases
    if n == 1:
        return m
    if n == 2:
        return m * (m - 1)
    
    # Initialize dp array
    dp = [0] * m
    for i in range(m):
        dp[i] = 1
    
    # Fill dp array
    for _ in range(2, n):
        new_dp = [0] * m
        for j in range(m):
            for k in range(m):
                if j != k:
                    new_dp[j] += dp[k]
                    new_dp[j] %= MOD
        dp = new_dp
    
    # Sum up all valid distributions
    result = sum(dp) % MOD
    return result

def mod_exp(base, exp, mod):
    result = 1
    base = base % mod
    while exp > 0:
        if (exp % 2) == 1:
            result = (result * base) % mod
        exp = exp >> 1
        base = (base * base) % mod
    return result

def distinct_adjacent(N, M):
    MOD = 998244353
    if N == 2:
        return (M * (M - 1)) % MOD
    
    # Calculate (M-1)^N
    power_result = mod_exp(M - 1, N, MOD)
    
    # Calculate (-1)^N
    sign = (-1) ** N
    
    # Calculate the final result
    result = (power_result + sign * (M - 1)) % MOD
