def max_alternating_subarray_sum(arr):
    if len(arr) == 1:
        return arr[0]
    
    max_odd = max_even = arr[0]
    for i in range(1, len(arr)):
        if arr[i] % 2 == 0:
            # If the current element is even, add it to the max_even subarray
            max_even = max(max_even + arr[i], arr[i])
            # The max_odd subarray ends at the previous even element
            max_odd = max(max_odd, max_even - arr[i])
        else:
            # If the current element is odd, add it to the max_odd subarray
            max_odd = max(max_odd + arr[i], arr[i])
            # The max_even subarray ends at the previous odd element
            max_even = max(max_even, max_odd - arr[i])
    
    return max(max_odd, max_even)
